% High-level architecture: layered, read-only ingestion; optional operator-initiated control path.
\begin{tikzpicture}[
  font=\small,
  box/.style={draw=accent, fill=accent!8, rounded corners=2pt, align=center, minimum height=0.9cm, text width=2.7cm},
  ext/.style={draw=black!50, fill=black!5, rounded corners=2pt, align=center, minimum height=0.9cm, text width=3.1cm},
  layer/.style={draw=black!25, dashed, rounded corners=4pt, inner sep=6pt},
  lbl/.style={font=\footnotesize\bfseries\color{black!60}, anchor=south west},
  flow/.style={-{Stealth[length=2.2mm]}, thick, accent},
  ctrl/.style={-{Stealth[length=2.2mm]}, thick, dashed, black!45},

]
% L0 sources
\node[ext] (stt) {STT software layer\\\footnotesize(database, logger, API)};
\node[ext, below=of stt] (sim) {BMS simulator};
\node[ext, below=of sim] (rep) {Prepared datasets\\\footnotesize(recorded runs)};

% L1 adapters
\node[box, right=1.1cm of stt] (a1) {STT adapter\\\footnotesize(read-only)};
\node[box, right=1.1cm of sim] (a2) {Simulator adapter};
\node[box, right=1.1cm of rep] (a3) {Replay adapter};

% L2 KPI service
\node[box, right=1.1cm of a1] (cal) {Calibration\\\footnotesize(STT logic, reused)};

\node[box, below=of cond] (kpi) {KPI computation\\\footnotesize(per-run streams)};

% L3 distribution and views
\node[box, right=1.1cm of cond] (ws) {Streaming\\interface};
\node[box, above=of ws] (demo) {Customer\\demonstration view};
\node[box, below=of ws] (lead) {Project-lead view};
\node[box, below=of lead] (eng) {Engineer\\monitoring view};

% Layers
\begin{scope}[on background layer]
\node[layer, fit=(a1)(a3)] (L1) {};
\node[layer, fit=(cal)(kpi)] (L2) {};
\node[layer, fit=(demo)(eng)(ws)] (L3) {};
\end{scope}
\node[lbl] at (L1.north west) {Ingestion adapters};
\node[lbl] at (L2.north west) {KPI service};
\node[lbl] at (L3.north west) {Distribution and views};

% Data flows (one way)
\draw[flow] (stt) -- (a1);
\draw[flow] (sim) -- (a2);
\draw[flow] (rep) -- (a3);
\draw[flow] (a1.east) -- (cal.west);
\draw[flow] (a2.east) -| ([xshift=-0.3cm]cal.west) -- (cal.west);
\draw[flow] (a3.east) -| ([xshift=-0.3cm]cal.west);
\draw[flow] (cal) -- (cond);
\draw[flow] (cond) -- (kpi);
\draw[flow] (kpi.east) -- ++(0.35,0) |- (ws.west);
\draw[flow] (ws) -- (demo);
\draw[flow] (ws) -- (lead);
\draw[flow] (ws.east) -- ++(0.25,0) |- (eng.east);

% Optional operator-initiated control path (human-in-the-loop)
\draw[ctrl] (demo.north) -- ++(0,1.15) -| node[pos=0.25, above, font=\footnotesize\color{black!60}] {optional operator-initiated command via the STT's own interface} (stt.north);
\end{tikzpicture}
